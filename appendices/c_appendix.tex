\chapter{Dataset}
\label{app:dataset}

This chapter reports the harm taxonomy and scenario allocation used to
construct the benchmark.

\section{Harm Taxonomy}
\label{app:crosswalk}

The taxonomy is based on the Online Safety Act 2023
\cite{OnlineSafetyAct2025} and supplemented by Body Image and Emotional
Dependency. Table~\ref{tab:crosswalk} compares its categories with prior
child-safety taxonomies, while Table~\ref{tab:allocation} gives the allocation
of the 200 benchmark scenarios.

\begin{statutecase}{Online Safety Act 2023, s.61}

\statfield{Primary Priority Content That Is Harmful to Children}

\begin{statblock}
\small\ttfamily
\slevi{(1)}{\enquote{Primary priority content that is harmful to children} means content of any of the following kinds.}
\slevi{(2)}{\statutory{Pornographic content, other than content within subsection (6).}}
\slevi{(3)}{\statutory{Content which encourages, promotes or provides instructions for suicide.}}
\slevi{(4)}{\statutory{Content which encourages, promotes or provides instructions for an act of deliberate self-injury.}}
\slevi{(5)}{\statutory{Content which encourages, promotes or provides instructions for an eating disorder or behaviours associated with an eating disorder.}}
\slevi{(6)}{Content is within this subsection if it}
\slevii{(a)}{consists only of text, or}
\slevii{(b)}{consists only of text accompanied by $[\ldots]$}
\slevi{(7)}{In this section and section 62 \enquote{injury} includes poisoning.}
\end{statblock}

\end{statutecase}

\begin{statutecase}{Online Safety Act 2023, s.62}

\statfield{Priority Content That Is Harmful to Children}

\begin{statblock}
\small\ttfamily
\slevi{(1)}{\enquote{Priority content that is harmful to children} means content of any of the following kinds.}
\slevi{(2)}{\statutory{Content which is abusive and which targets any of the following characteristics}}
\slevii{(a)}{race,}
\slevii{(b)}{religion,}
\slevii{(c)}{sex,}
\slevii{(d)}{sexual orientation,}
\slevii{(e)}{disability, or}
\slevii{(f)}{gender reassignment.}
\slevi{(3)}{\statutory{Content which incites hatred against people $[\ldots]$}}
\slevi{(4)}{\statutory{Content which encourages, promotes or provides instructions for an act of serious violence against a person.}}
\slevi{(5)}{\statutory{Bullying content.}}
\slevi{(6)}{Content which}
\slevii{(a)}{depicts real or realistic serious violence against a person;}
\slevii{(b)}{depicts the real or realistic serious injury of a person in graphic detail.}
\slevi{(7)}{Content which}
\slevii{(a)}{depicts real or realistic serious violence against an animal;}
\slevii{(b)}{depicts the real or realistic serious injury of an animal in graphic detail;}
\slevii{(c)}{realistically depicts serious violence against a fictional creature or the serious injury of a fictional creature in graphic detail.}
\slevi{(8)}{\statutory{Content which encourages, promotes or provides instructions for a challenge or stunt highly likely to result in serious injury to the person who does it or to someone else.}}
\slevi{(9)}{\statutory{Content which encourages a person to ingest, inject, inhale or in any other way self-administer}}
\slevii{(a)}{a physically harmful substance;}
\slevii{(b)}{a substance in such a quantity as to be physically harmful.}
\slevi{(12)}{For the purposes of subsection (5) content may, in particular, be \enquote{bullying content} if it is content targeted against a person which}
\slevii{(a)}{conveys a serious threat;}
\slevii{(b)}{is humiliating or degrading;}
\slevii{(c)}{forms part of a campaign of mistreatment.}
\slevi{(13)}{In subsection (6) \enquote{person} is not limited to a real person.}
\slevi{(14)}{In subsection (7) \enquote{animal} is not limited to a real animal.}
\end{statblock}

\end{statutecase}
